package com.university.newsapp

import android.content.Intent
import android.os.Bundle
import android.view.View
import android.widget.LinearLayout
import android.widget.ProgressBar
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import com.google.android.material.snackbar.Snackbar
import com.university.newsapp.model.Comment
import com.university.newsapp.model.Post
import com.university.newsapp.model.User
import com.university.newsapp.network.RetrofitClient
import kotlinx.coroutines.launch

class PostDetailActivity : AppCompatActivity() {

    private lateinit var post: Post
    private lateinit var tvTitle: TextView
    private lateinit var tvBody: TextView
    private lateinit var tvAuthorName: TextView
    private lateinit var tvAuthorEmail: TextView
    private lateinit var tvAuthorCompany: TextView
    private lateinit var containerComments: LinearLayout
    private lateinit var progressBar: ProgressBar

    private var author: User? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_post_detail)

        supportActionBar?.setDisplayHomeAsUpEnabled(true)
        supportActionBar?.title = "Post Detail"

        // Get post from intent
        post = intent.getSerializableExtra("post") as? Post ?: return

        // Initialize UI
        initializeUI()

        // Load details
        loadPostDetails()
    }

    override fun onSupportNavigateUp(): Boolean {
        finish()
        return true
    }

    private fun initializeUI() {
        tvTitle = findViewById(R.id.tvPostTitle)
        tvBody = findViewById(R.id.tvPostBody)
        tvAuthorName = findViewById(R.id.tvAuthorName)
        tvAuthorEmail = findViewById(R.id.tvAuthorEmail)
        tvAuthorCompany = findViewById(R.id.tvAuthorCompany)
        containerComments = findViewById(R.id.containerComments)
        progressBar = findViewById(R.id.progressBar)

        // Set post content
        tvTitle.text = post.title
        tvBody.text = post.body
    }

    private fun loadPostDetails() {
        lifecycleScope.launch {
            try {
                // Load author
                author = RetrofitClient.apiService.getUserById(post.userId)
                displayAuthor()

                // Load comments
                val comments = RetrofitClient.apiService.getCommentsByPost(post.id)
                displayComments(comments)

                progressBar.visibility = ProgressBar.GONE
            } catch (e: Exception) {
                Snackbar.make(
                    tvTitle,
                    "Failed to load details: ${e.message}",
                    Snackbar.LENGTH_SHORT
                ).show()
            }
        }
    }

    private fun displayAuthor() {
        if (author != null) {
            tvAuthorName.text = author!!.name
            tvAuthorEmail.text = author!!.email
            tvAuthorCompany.text = author!!.company.name

            // Make author section clickable
            findViewById<android.widget.CardView>(R.id.cardAuthor).setOnClickListener {
                val intent = Intent(this, UserProfileActivity::class.java)
                intent.putExtra("user", author)
                startActivity(intent)
            }
        }
    }

    private fun displayComments(comments: List<Comment>) {
        if (comments.isEmpty()) {
            val tv = TextView(this).apply {
                text = "No comments yet"
                setPadding(16, 16, 16, 16)
            }
            containerComments.addView(tv)
            return
        }

        comments.forEach { comment ->
            val commentView = createCommentView(comment)
            containerComments.addView(commentView)
        }
    }

    private fun createCommentView(comment: Comment): android.widget.CardView {
        val cardView = android.widget.CardView(this).apply {
            layoutParams = LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.MATCH_PARENT,
                LinearLayout.LayoutParams.WRAP_CONTENT
            ).apply {
                setMargins(0, 8, 0, 8)
            }
            radius = 8f
            cardElevation = 2f
        }

        val layout = LinearLayout(this).apply {
            layoutParams = android.widget.CardView.LayoutParams(
                android.widget.CardView.LayoutParams.MATCH_PARENT,
                android.widget.CardView.LayoutParams.WRAP_CONTENT
            )
            orientation = LinearLayout.VERTICAL
            setPadding(16, 16, 16, 16)
        }

        val nameView = TextView(this).apply {
            text = comment.name
            textSize = 14f
            typeface = android.graphics.Typeface.create(android.graphics.Typeface.DEFAULT, android.graphics.Typeface.BOLD)
        }

        val emailView = TextView(this).apply {
            text = comment.email
            textSize = 12f
            setTextColor(android.graphics.Color.GRAY)
            layoutParams = LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.MATCH_PARENT,
                LinearLayout.LayoutParams.WRAP_CONTENT
            ).apply { topMargin = 4 }
        }

        val bodyView = TextView(this).apply {
            text = comment.body
            textSize = 14f
            layoutParams = LinearLayout.LayoutParams(
                LinearLayout.LayoutParams.MATCH_PARENT,
                LinearLayout.LayoutParams.WRAP_CONTENT
            ).apply { topMargin = 8 }
        }

        layout.addView(nameView)
        layout.addView(emailView)
        layout.addView(bodyView)
        cardView.addView(layout)

        return cardView
    }
}
